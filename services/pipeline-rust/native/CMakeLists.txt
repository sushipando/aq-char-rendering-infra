cmake_minimum_required(VERSION 3.22)
project(aqw_avif_rgba C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(AVIF_CODEC_AOM LOCAL CACHE STRING "" FORCE)
set(AVIF_CODEC_DAV1D OFF CACHE STRING "" FORCE)
set(AVIF_LIBYUV OFF CACHE STRING "" FORCE)
set(AVIF_BUILD_APPS OFF CACHE BOOL "" FORCE)
set(AVIF_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${LIBAVIF_SOURCE}" libavif)
add_executable(avif-rgba avif_rgba.c)
target_link_libraries(avif-rgba PRIVATE avif)
# Keep the Lambda helper independent of runtime C++ library packaging.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(avif-rgba PRIVATE -static-libstdc++ -static-libgcc)
endif()
